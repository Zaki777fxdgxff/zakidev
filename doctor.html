'owner' : (d.blackBadge ? 'black' : (d.verified ? 'normal' : 'none'))); document.getElementById('chatRoomAvatar').src = d.avatar || DEFAULT_AVATAR; var nameHtml = escapeHtml(currentChatPartnerName); if (currentChatPartnerBadge !== 'none') { nameHtml += ' '; } document.getElementById('chatRoomName').innerHTML = nameHtml; document.getElementById('chatRoomUsername').textContent = '@' + (d.username || 'user'); } var chatId = [currentUserKey, partnerKey].sort().join('__'); applyChatBackground(currentBgSelection); var msgsRef = fb.collection(fb.db, 'chats', chatId, 'messages'); if (chatUnsubscribe) chatUnsubscribe(); chatUnsubscribe = fb.onSnapshot( fb.query(msgsRef, fb.orderBy('createdAt', 'asc')), function(snapshot) { var box = document.getElementById('chatMessages'); box.innerHTML = ''; if (snapshot.empty) { box.innerHTML = '
' + t('start_chat') + '
'; return; } snapshot.forEach(function(ds) { var m = ds.data(); var isMe = m.from === currentUserKey; var row = document.createElement('div'); row.className = 'chat-msg-row ' + (isMe ? 'me' : 'other'); var bubble = document.createElement('div'); bubble.className = 'chat-msg ' + (isMe ? 'me' : 'other'); var timeStr = ''; if (m.createdAt) { try { timeStr = new Date(m.createdAt).toLocaleTimeString(localStorage.getItem('lang') === 'en' ? 'en-US' : 'ar-EG', {hour:'2-digit', minute:'2-digit'}); } catch(e){} } bubble.innerHTML = escapeHtml(m.text || '') + (timeStr ? '' + timeStr + '' : ''); row.appendChild(bubble); box.appendChild(row); }); box.scrollTop = box.scrollHeight; markChatAsRead(chatId); } ); } catch (e) { console.error(e); toast('خطأ: ' + e.message); } }; // 3. جلب حالة الأزرار عند فتح صفحة التعديل var originalShowEditProfile = window.showEditProfile; window.showEditProfile = function() { if(originalShowEditProfile) originalShowEditProfile(); if (currentUserKey) { getFB().then(fb => { fb.getDoc(fb.doc(fb.db, 'profiles', currentUserKey)).then(snap => { if(snap.exists()) { var d = snap.data(); if(document.getElementById('pfHideFromSearch')) document.getElementById('pfHideFromSearch').checked = d.hideFromSearch || false; if(document.getElementById('pfDisableChat')) document.getElementById('pfDisableChat').checked = d.disableChat || false; } }); }); } };
